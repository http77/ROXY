Product Box #${boxCount}
`; container.appendChild(boxHtml); }; window.removeBulkBox = (id) => { const el = document.getElementById(`bulk-box-${id}`); if (el) el.remove(); }; // Handle Bulk Upload Click const bulkBtn = document.getElementById("bulkUploadBtn"); if (bulkBtn) { bulkBtn.addEventListener("click", async () => { const boxes = document.querySelectorAll(".bulk-item"); if (boxes.length === 0) { alert("Please add at least one product box!"); return; } const productsToUpload = []; for (let box of boxes) { const name = box.querySelector(".bulk-name").value.trim(); const category = box.querySelector(".bulk-cat").value; const price = Number(box.querySelector(".bulk-price").value); const img = box.querySelector(".bulk-img").value.trim(); if (!name || !price || !img) { alert("সব বক্সের নাম, দাম এবং ইমেজ লিংক সঠিকভাবে পূরণ করুন!"); return; } productsToUpload.push({ name, category, price, img, stockStatus: "In Stock" }); } try { for (let p of productsToUpload) { await addDoc(collection(db, "products"), p); } alert(`Successfully uploaded ${productsToUpload.length} products!`); document.getElementById("bulkContainer").innerHTML = ""; boxCount = 0; addBulkBox(); // Reset with 1 empty box if (typeof loadProducts === "function") loadProducts(); } catch (err) { console.error(err); alert("Error uploading products: " + err.message); } }); } // Initialize with 1 default box on load addBulkBox(); ```[cite: 6] এখন আপনার অ্যাডমিন প্যানেলের আগের ডিজাইন, অর্ডার লিস্ট এবং অন্যান্য সব ফিচার একদম অক্ষুণ্ণ থাকবে, এবং বাল্ক আপলোড ট্যাবে গিয়ে আপনি ইচ্ছামতো বক্স বাড়িয়ে একসাথে একাধিক প্রোডাক্ট খুব সহজে আপলোড করতে পারবেন[cite: 5, 6]।
